#     Verify: python -c "import torch; print(torch.cuda.is_available())"  -> True)
# 2) pip install -r requirements-airllm.txt
# 3) python scripts/airllm_setup.py             (shows the model this machine should use)
#    python scripts/airllm_setup.py --download  (downloads + splits it: tens of GB)
